async function openPrivate(peer) { const ids = [user.uid, peer.uid].sort(); const cid = ids.join('_'); const ref = doc(db, 'conversations', cid); const snapshot = await getDoc(ref); if (!snapshot.exists()) { await setDoc(ref, { members: ids, createdAt: serverTimestamp(), lastMessage: '', updatedAt: serverTimestamp() }); } active = cid; activePeer = peer; $('title').textContent = peer.displayName || 'Private chat'; $('subtitle').textContent = 'Private conversation'; $('chat-avatar').textContent = peer.memoji || '🙂'; startChat(); changeView('chat'); $('sidebar').classList.remove('open'); }
